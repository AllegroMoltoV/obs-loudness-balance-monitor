cmake_minimum_required(VERSION 3.20)
project(preview_probe LANGUAGES CXX)
enable_testing()
add_executable(preview-core core.cpp)
target_compile_features(preview-core PRIVATE cxx_std_17)
add_test(NAME recording-core COMMAND preview-core)
if(APPLE OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
 add_executable(lbm-preview main.cpp)
 target_compile_features(lbm-preview PRIVATE cxx_std_17)
 target_compile_options(lbm-preview PRIVATE -Wall -Wextra -Wpedantic)
 if(APPLE)
  target_sources(lbm-preview PRIVATE playback-macos.cpp)
  target_link_libraries(lbm-preview PRIVATE "-framework AudioToolbox" "-framework CoreAudio" "-framework CoreFoundation")
 else()
  find_package(Threads REQUIRED)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(PULSE REQUIRED IMPORTED_TARGET libpulse)
  target_sources(lbm-preview PRIVATE playback-pulse.cpp)
  target_link_libraries(lbm-preview PRIVATE PkgConfig::PULSE Threads::Threads)
 endif()
endif()
